name=Host Arduino Core
version=1.4.0

compiler.path=
compiler.c.cmd=gcc
compiler.cpp.cmd=g++
compiler.ar.cmd=ar
compiler.size.cmd=size
compiler.ar.flags=rcs
compiler.ar.extra_flags=

compiler.c.flags=-O2 -Wall -ffunction-sections -fdata-sections
compiler.c.flags.windows=-O2 -Wall -ffunction-sections -fdata-sections -IC:/msys64/ucrt64/include
compiler.cxx.flags=-O2 -Wall -ffunction-sections -fdata-sections
compiler.cxx.flags.windows=-O2 -Wall -ffunction-sections -fdata-sections -IC:/msys64/ucrt64/include
compiler.ldflags=
compiler.ldflags.windows=-LC:/msys64/ucrt64/lib
# Core-controlled defines, accumulated from board-menu options. Kept
# separate from `build.extra_flags` so that user-side overrides (e.g.
# arduino-cli's `--build-property build.extra_flags=...`, which is what
# `build_config.toml` in pytest-embedded-arduino-cli emits to inject
# things like WIFI_SSID) compose cleanly instead of clobbering our menu
# selections.
build.core_flags=-DHOST_ARDUINO=1 {build.extra_flags_mode} {build.extra_flags_tls} {build.extra_flags_m5gfx_board} {build.extra_flags_m5gfx_scale} {build.extra_flags_m5gfx_rotation}
build.extra_flags=

recipe.preproc.macros={compiler.path}{compiler.cpp.cmd} -x c++ -E -CC {compiler.cxx.flags} {build.core_flags} {build.extra_flags} {includes} -I{runtime.platform.path}/cores/{build.core} -I{build.path}/sketch {source_file} -o {preprocessed_file_path}
recipe.c.o.pattern={compiler.path}{compiler.c.cmd} {compiler.c.flags} {build.core_flags} {build.extra_flags} {includes} -I{runtime.platform.path}/cores/{build.core} -I{build.path}/sketch -c -o {object_file} {source_file}
recipe.cpp.o.pattern={compiler.path}{compiler.cpp.cmd} {compiler.cxx.flags} {build.core_flags} {build.extra_flags} {includes} -I{runtime.platform.path}/cores/{build.core} -I{build.path}/sketch -c -o {object_file} {source_file}
recipe.ar.pattern={compiler.path}{compiler.ar.cmd} {compiler.ar.flags} {compiler.ar.extra_flags} "{archive_file_path}" "{object_file}"

compiler.c.elf.cmd={compiler.cpp.cmd}
compiler.c.elf.flags={compiler.cxx.flags}
recipe.c.combine.pattern={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} {object_files} {archive_file_path} {build.core_flags} {build.extra_flags} -pthread -o {build.path}/{build.project_name}.out
recipe.cpp.combine.pattern={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} {object_files} {archive_file_path} {build.core_flags} {build.extra_flags} -pthread -o {build.path}/{build.project_name}.out
recipe.c.combine.pattern.windows={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} {object_files} {archive_file_path} {build.core_flags} {build.extra_flags} -lws2_32 -o {build.path}/{build.project_name}.out
recipe.cpp.combine.pattern.windows={compiler.path}{compiler.c.elf.cmd} {compiler.c.elf.flags} {compiler.ldflags} {object_files} {archive_file_path} {build.core_flags} {build.extra_flags} -lws2_32 -o {build.path}/{build.project_name}.out

tools.run.upload.pattern="{build.path}/{build.project_name}.out"
tools.run.upload.pattern.windows="{build.path}/{build.project_name}.out"

recipe.size.pattern={compiler.path}{compiler.size.cmd} {build.path}/{build.project_name}.out

compiler.core=core
build.core=host
